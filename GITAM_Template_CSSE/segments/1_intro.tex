\begin{frame}{Introduction: Problem Context \& Approach}
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    {\large\textbf{\textcolor{MyUnivGreen}{Problem Context}}}\\[6pt]
    \begin{itemize}
      \item Visually impaired individuals face acute daily barriers in unstructured Indian road environments.
      \item Traditional white canes only offer physical contact obstacle detection with zero contextual awareness.
      \item Existing electronic aids require bulky, expensive custom hardware (LiDAR, ultrasonic vests).
    \end{itemize}
  \end{column}
  
  \begin{column}{0.48\textwidth}
    {\large\textbf{\textcolor{MyUnivGreen}{Smartphone-Based Approach}}}\\[6pt]
    \begin{itemize}
      \item Low-cost solution utilizing the user's commercial smartphone.
      \item Real-time YOLO computer vision for obstacle detection (vehicles, pedestrians, potholes).
      \item Gemini Vision-Language Model for contextual scene understanding.
      \item Audio-first interface via turn-by-turn GPS, OCR sign reading, and speech synthesis (STT/TTS).
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Real-World Situations Addressed}
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    {\large\textbf{\textcolor{MyUnivGreen}{Road Safety \& Mobility}}}\\[6pt]
    \begin{itemize}
      \item \textbf{Crossing Roads}: Approach speed and directional alerts.
      \item \textbf{Walking on Footpaths}: Pedestrians, poles, uneven terrain.
      \item \textbf{Potholes}: Depth and surface hazard recognition.
      \item \textbf{Fast Vehicles}: Approaching motorcycle warnings.
      \item \textbf{Crowded Streets}: Real-time navigation in markets.
    \end{itemize}
  \end{column}
  
  \begin{column}{0.48\textwidth}
    {\large\textbf{\textcolor{MyUnivGreen}{Wayfinding \& Accessibility}}}\\[6pt]
    \begin{itemize}
      \item \textbf{Bus Stops}: GPS and camera-guided transit locating.
      \item \textbf{Road Signs}: OCR text extraction with spoken output.
      \item \textbf{Blocked Paths}: Parked vehicles, construction debris, vendors.
      \item \textbf{Key Destinations}: Audio routing to clinics and pharmacies.
      \item \textbf{Emergency Help}: One-touch SOS with live GPS sharing.
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}
