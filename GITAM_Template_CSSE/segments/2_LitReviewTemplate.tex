% Paper 1: Senjam et al. (Clinical Ophthalmology, 2021)
\begin{frame}{Literature Review: Smartphones-Based Assistive Technology}
{\large\textbf{\textcolor{MyUnivGreen}{S. S. Senjam et al. (Clinical Ophthalmology, 2021) \cite{senjam2021smartphones}}}}\\[6pt]

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{\textcolor{MyUnivGreen}{Contributions \& Methodology}}\\[4pt]
    \begin{itemize}
      \item Evaluated smartphone accessibility apps (Seeing AI, Google Lookout).
      \item Investigated computer vision and sensor fusion on mobile hardware.
      \item Evaluated audio/speech interface for non-visual environmental feedback.
      \item Demonstrated $>70\%$ improvement in daily task autonomy.
    \end{itemize}
  \end{column}
  
  \begin{column}{0.48\textwidth}
    \textbf{\textcolor{MyUnivGreen}{Inputs, Outputs \& Limitations}}\\[4pt]
    \begin{itemize}
      \item \textbf{Input:} Mobile camera stream, touch gestures, voice input.
      \item \textbf{Output:} Synthesized speech and haptic notifications.
      \item \textbf{Limitation 1:} Limited real-time navigation on Indian roads.
      \item \textbf{Limitation 2:} Heavy dependency on continuous cloud connectivity.
      \item \textbf{Limitation 3:} Lacks semantic scene reasoning for dynamic hazards.
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}

% Paper 2: Kathiria et al. (Neurocomputing, 2024)
\begin{frame}{Literature Review: Advancements in Assistive Systems}
{\large\textbf{\textcolor{MyUnivGreen}{P. Kathiria et al. (Neurocomputing, 2024) \cite{kathiria2024assistive}}}}\\[6pt]

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{\textcolor{MyUnivGreen}{Contributions \& Methodology}}\\[4pt]
    \begin{itemize}
      \item Benchmarked sensors: LiDAR vs. Ultrasound vs. Camera vision.
      \item Surveyed deep learning detectors (YOLO, SSD, Faster R-CNN).
      \item Evaluated multimodal integration (RGB-D + IMU + GPS).
      \item Vision-based systems outperform ultrasound in hazard recognition.
    \end{itemize}
  \end{column}
  
  \begin{column}{0.48\textwidth}
    \textbf{\textcolor{MyUnivGreen}{Inputs, Outputs \& Limitations}}\\[4pt]
    \begin{itemize}
      \item \textbf{Input:} Video frames, depth maps, GPS coordinates.
      \item \textbf{Output:} Directional spatial audio, speech synthesis.
      \item \textbf{Limitation 1:} Most reviewed systems require dedicated sensor rigs.
      \item \textbf{Limitation 2:} High cost ($>\$500$) prohibitive for mass adoption.
      \item \textbf{Limitation 3:} Lack of unified end-to-end smartphone software.
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}

% Paper 3: Manirajee et al. (SLR, 2024)
\begin{frame}{Literature Review: Systematic Review of Assistive Technologies}
{\large\textbf{\textcolor{MyUnivGreen}{L. Manirajee et al. (IJARBSS, 2024) \cite{manirajee2024assistive}}}}\\[6pt]

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{\textcolor{MyUnivGreen}{Contributions \& Methodology}}\\[4pt]
    \begin{itemize}
      \item PRISMA systematic review mapping 50+ assistive publications.
      \item Analyzed cost vs. usability barrier across user demographics.
      \item Concluded smartphones are the most scalable assistive medium.
      \item Highlighted strong user preference for hands-free audio guidance.
    \end{itemize}
  \end{column}
  
  \begin{column}{0.48\textwidth}
    \textbf{\textcolor{MyUnivGreen}{Inputs, Outputs \& Limitations}}\\[4pt]
    \begin{itemize}
      \item \textbf{Input:} User mobility datasets and navigation metrics.
      \item \textbf{Output:} Comparative usability matrices and gap analysis.
      \item \textbf{Limitation 1:} Few existing works address developing nation roads.
      \item \textbf{Limitation 2:} No conversational AI integration for scene querying.
      \item \textbf{Limitation 3:} Existing tools lack offline edge inference capability.
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}
