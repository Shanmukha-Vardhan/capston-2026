\begin{frame}{Project Objectives}
{\large\textbf{\textcolor{MyUnivGreen}{Core Research \& Development Objectives}}}\\[8pt]

\begin{enumerate}
    \setlength\itemsep{8pt}
    \item \textbf{Develop a Cross-Platform Mobile Application:}
    Build an accessible smartphone application (React Native/Expo) delivering real-time assistance directly on commercial smartphones, eliminating specialized hardware costs.
    
    \item \textbf{Integrate AI Object Detection \& Contextual Scene Understanding:}
    Deploy an optimized YOLO edge model for real-time hazard detection (vehicles, pedestrians, potholes) alongside Gemini Vision-Language Models for deep semantic environmental description.
    
    \item \textbf{Enable Voice-Guided GPS Navigation \& OCR Reading:}
    Incorporate turn-by-turn auditory routing via GPS, OCR capability for reading signs and transit boards aloud, and a conversational hands-free voice interface (STT/TTS).
\end{enumerate}
\end{frame}
